Decorators

import time

def timer(func):
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} took {end-start:.4f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "Done"

slow_function()  # slow_function took 1.00xxs

Decorators wrap functions to add behavior without modifying original code.

Generators (yield)

# Generator function - yields values one at a time
def fibonacci(n):
    a, b = 0, 1
    for _ in range(n):
        yield a
        a, b = b, a + b

# Use generator
fib = fibonacci(8)
print(list(fib))  # [0, 1, 1, 2, 3, 5, 8, 13]

# Generator expression (like list comp but lazy)
squares = (x**2 for x in range(5))
print(next(squares))  # 0
print(next(squares))  # 1

Generators use yield to produce values lazily — memory efficient for large data.

Generators yield se values lazily produce karte hain — bade data ke liye memory efficient.

Recursion

# Factorial using recursion
def factorial(n):
    if n <= 1:       # base case
        return 1
    return n * factorial(n - 1)  # recursive call

print(factorial(5))  # 120 (5*4*3*2*1)

# Fibonacci using recursion
def fib(n):
    if n <= 1: return n
    return fib(n-1) + fib(n-2)

print(fib(7))  # 13

Recursion: function calls itself. Must have base case to stop infinite calls.

Recursion: function khud ko call karta hai. Base case zaroori hai infinite calls rokne ke liye.

Class vs Instance vs Static Methods

class MathUtils:
    pi = 3.14159  # class attribute
    
    def __init__(self, value):
        self.value = value  # instance attribute
    
    def double(self):  # instance method
        return self.value * 2
    
    @classmethod
    def circle_area(cls, r):  # class method
        return cls.pi * r ** 2
    
    @staticmethod
    def add(a, b):  # static method
        return a + b

obj = MathUtils(5)
print(obj.double())            # 10
print(MathUtils.circle_area(7))  # 153.93
print(MathUtils.add(3, 4))    # 7

Instance method uses self, classmethod uses cls, staticmethod needs neither.

Instance method self use karta hai, classmethod cls, staticmethod koi bhi nahi.

@property Decorator

class Circle:
    def __init__(self, radius):
        self._radius = radius
    
    @property
    def radius(self):
        return self._radius
    
    @radius.setter
    def radius(self, value):
        if value < 0:
            raise ValueError("Radius can't be negative")
        self._radius = value
    
    @property
    def area(self):
        return 3.14159 * self._radius ** 2

c = Circle(5)
print(c.radius)  # 5 (getter)
print(c.area)    # 78.54 (computed property)
c.radius = 10    # setter
# c.radius = -1  # raises ValueError

@property creates getter/setter with validation — access like attribute, runs like method.

Inheritance (Single, Multiple, Multilevel) & super()

# Single Inheritance
class Animal:
    def __init__(self, name):
        self.name = name
    def speak(self):
        return "..."

class Dog(Animal):
    def speak(self):
        return f"{self.name} says Woof!"

# Multilevel Inheritance
class Puppy(Dog):

        return f"{self.name} is playing!"

# Multiple Inheritance
class Flyable:
    def fly(self): return "Flying!"

class Swimmable:
    def swim(self): return "Swimming!"

class Duck(Flyable, Swimmable):
    pass

Single: one parent. Multiple: many parents. Multilevel: chain. super() calls parent method.

Single: ek parent. Multiple: kai parents. Multilevel: chain. super() parent method call karta hai.

Polymorphism & Method Overriding

class Shape:
    def area(self):
        raise NotImplementedError

class Rectangle(Shape):
    def __init__(self, w, h):
        self.w, self.h = w, h
    def area(self):
        return self.w * self.h

class Circle(Shape):
    def __init__(self, r):
        self.r = r
    def area(self):
        return 3.14 * self.r ** 2

# Same method name, different behavior
shapes = [Rectangle(4, 5), Circle(3)]
for s in shapes:
    print(f"Area: {s.area()}")
# Area: 20, Area: 28.26

Polymorphism: same method name behaves differently in different classes (overriding).

Polymorphism: same method name alag classes mein alag behave karta hai (overriding).

Encapsulation (public / _protected / __private)

class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner          # public
        self._bank = "SBI"          # protected (convention)
        self.__balance = balance    # private (name mangling)
    
    def deposit(self, amount):
        self.__balance += amount
    
    def get_balance(self):
        return self.__balance

acc = BankAccount("Manoj", 1000)
print(acc.owner)        # Manoj (public - accessible)
print(acc._bank)        # SBI (protected - accessible but shouldn't)
# print(acc.__balance)  # AttributeError! (private)
print(acc.get_balance())  # 1000 (via method)
# Name mangling: acc._BankAccount__balance works but shouldn't use

Public: accessible everywhere. _Protected: convention only. __Private: name mangling prevents direct access.

Public: sab jagah accessible. _Protected: sirf convention. __Private: name mangling direct access rokta hai.

Abstraction (ABC)

from abc import ABC, abstractmethod

class Vehicle(ABC):
    @abstractmethod
    def start(self):
        pass
    
    @abstractmethod
    def stop(self):
        pass
    
    def fuel_type(self):  # concrete method
        return "Petrol"

class Car(Vehicle):
    def start(self):
        return "Car started with key"
    def stop(self):
        return "Car stopped"

# v = Vehicle()  # TypeError! Can't instantiate ABC
c = Car()
print(c.start())      # Car started with key
print(c.fuel_type())  # Petrol

ABC forces subclasses to implement abstract methods. Can't instantiate ABC directly.

ABC subclasses ko abstract methods implement karne pe majboor karta hai. ABC directly instantiate nahi ho sakta.

Magic/Dunder Methods (__str__, __add__, __len__, __eq__)

class Vector:
    def __init__(self, x, y):
        self.x, self.y = x, y
    
    def __str__(self):
        return f"Vector({self.x}, {self.y})"
    
    def __repr__(self):
        return f"Vector({self.x}, {self.y})"
    
    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)
    
    def __len__(self):
        return int((self.x**2 + self.y**2)**0.5)
    
    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

v1 = Vector(3, 4)
v2 = Vector(1, 2)
print(v1)           # Vector(3, 4) — __str__
print(v1 + v2)      # Vector(4, 6) — __add__
print(len(v1))      # 5 — __len__
print(v1 == v2)     # False — __eq__

Magic methods customize operators (+, ==, len, print) for your objects.

Magic methods aapke objects ke liye operators (+, ==, len, print) customize karte hain.

File Handling (open/read/write/with)

# Write to file
with open("data.txt", "w") as f:
    f.write("Line 1\n")
    f.write("Line 2\n")

# Read entire file
with open("data.txt", "r") as f:
    content = f.read()
    print(content)

# Read line by line
with open("data.txt", "r") as f:
    for line in f:
        print(line.strip())

# Append to file
with open("data.txt", "a") as f:
    f.write("Line 3\n")

# Read all lines as list
with open("data.txt") as f:
    lines = f.readlines()  # ['Line 1\n', 'Line 2\n', ...]

Use 'with' for safe file handling. Modes: r(read), w(write), a(append), r+(read+write).

Exception Handling (try/except/finally/custom)

# Basic try/except/else/finally
try:
    result = 10 / 0
except ZeroDivisionError as e:
    print(f"Error: {e}")
except (TypeError, ValueError) as e:
    print(f"Type/Value Error: {e}")
else:
    print("Success!")  # runs if no exception
finally:
    print("Always runs")  # cleanup code

# Custom Exception
class InsufficientFundsError(Exception):
    def __init__(self, balance, amount):
        self.message = f"Balance {balance}, tried {amount}"
        super().__init__(self.message)

def withdraw(balance, amount):
    if amount > balance:
        raise InsufficientFundsError(balance, amount)
    return balance - amount

try:
    withdraw(100, 200)
except InsufficientFundsError as e:
    print(e)  # Balance 100, tried 200

try catches errors, except handles them, finally always runs. raise throws custom exceptions.

try errors catch karta hai, except handle karta hai, finally hamesha chalta hai. raise custom exceptions throw karta hai.

Regex (re module)

import re

text = "Contact: manoj@college.edu, Phone: 9876543210"

# Search - find first match
email = re.search(r'[\w.]+@[\w.]+', text)
print(email.group())  # manoj@college.edu

# Findall - find all matches
numbers = re.findall(r'\d+', text)
print(numbers)  # ['9876543210']

# Sub - replace
clean = re.sub(r'\d+', 'XXXX', text)
print(clean)  # Contact: manoj@college.edu, Phone: XXXX

# Match - checks start of string
print(bool(re.match(r'Contact', text)))  # True

# Common patterns
# \d = digit, \w = word char, \s = space
# + = 1 or more, * = 0 or more, ? = 0 or 1
# ^ = start, $ = end, [] = character class

re module for pattern matching: search, findall, sub, match. \d=digit, \w=word, \s=space.

re module pattern matching ke liye: search, findall, sub, match. \d=digit, \w=word, \s=space.

Context Managers (with statement)

# Built-in context manager (file)
with open("test.txt", "w") as f:
    f.write("Hello")
# File auto-closed after 'with' block

# Custom context manager using class
class Timer:
    def __enter__(self):
        import time
        self.start = time.time()
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        import time
        print(f"Elapsed: {time.time()-self.start:.4f}s")
        return False  # don't suppress exceptions

with Timer():
    total = sum(range(1000000))

# Using contextlib decorator
from contextlib import contextmanager

@contextmanager
def managed_resource(name):
    print(f"Opening {name}")
    yield name  # value given to 'as' variable
    print(f"Closing {name}")

with managed_resource("DB") as r:
    print(f"Using {r}")
# Opening DB → Using DB → Closing DB

Context managers handle setup/cleanup automatically. Use __enter__/__exit__ or @contextmanager.

Context managers setup/cleanup automatically handle karte hain. __enter__/__exit__ ya @contextmanager use karo.
